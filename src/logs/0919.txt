def append_new_trait(self, trait_text: str):
    # 1. 通过 LLM 将文本转换为结构化节点
    normalized_node = self.trait_factory.normalize_new_chunk(...)
    
    # 2. 提取实体作为 Key (假设取第一个主实体)
    entities = normalized_node.get("linked_entities", [])
    if not entities: return "no_entity_found"
    
    primary_key = entities[0] 
    
    # 3. 执行合并逻辑
    if primary_key in self.trait_map:
        existing_trait = self.trait_map[primary_key]
        
        # 更新权重：取最大值 (代表最强烈的认知)
        existing_trait['emotional_weight'] = max(
            existing_trait['emotional_weight'], 
            normalized_node['emotional_weight']
        )
        
        # 检查冲突：如果行为模式改变，视为特质演进
        if existing_trait['action_mode'] != normalized_node['action_mode']:
            logger.info(f"Trait Shift detected for {primary_key}!")
            existing_trait['action_mode'] = normalized_node['action_mode']
            
        # 增加权重/频次计数
        existing_trait['mention_count'] = existing_trait.get('mention_count', 1) + 1
    else:
        # 新特质，直接存入
        normalized_node['mention_count'] = 1
        self.trait_map[primary_key] = normalized_node
